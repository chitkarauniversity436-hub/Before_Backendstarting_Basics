const http = require('http');

http.createServer((req,res)=>{
    // first you have to set the Type of response we are sending to the client
    res.setHeader('Content-Type','text/html');

    // then write the response to the client
    res.write("<h1>Heading 1</h1>");
    
    res.end();
}).listen(3000)




4) for Example
const data = [ {name:"tarun" , age:20} , {..} , {..}] ;  // array of object  

const http = require('http');

http.createServer((req,res)=>{
    res.writeHead("Content-Type","application/json");
    res.write(JSON.stringify(data));
    res.end();  
  
}).listen(3000)

// JSON.stringify(data) converts the JavaScript object/array into a JSON string so it can be sent as an HTTP response.

//Parse = convert the string back into an object
